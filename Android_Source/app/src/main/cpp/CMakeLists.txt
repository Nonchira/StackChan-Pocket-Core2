cmake_minimum_required(VERSION 3.22.1)
project(pocket_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Keep build-machine paths out of native assertion strings and debug metadata.
get_filename_component(POCKET_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_compile_options("-ffile-prefix-map=${POCKET_SOURCE_ROOT}=.")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
# Optional portable Vulkan dependencies; CPU-only builds need no host compiler.
if(DEFINED POCKET_VULKAN_ROOT AND NOT POCKET_VULKAN_ROOT STREQUAL "")
    file(TO_CMAKE_PATH "${POCKET_VULKAN_ROOT}" POCKET_VULKAN_ROOT)
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
    set(Vulkan_INCLUDE_DIR "${POCKET_VULKAN_ROOT}/Vulkan-Headers/include" CACHE PATH "" FORCE)
    set(Vulkan_GLSLC_EXECUTABLE "${ANDROID_NDK}/shader-tools/windows-x86_64/glslc.exe" CACHE FILEPATH "" FORCE)
    set(SPIRV-Headers_DIR "${POCKET_VULKAN_ROOT}/spirv-install/share/cmake/SPIRV-Headers" CACHE PATH "" FORCE)
    set(GGML_VULKAN_SHADERS_GEN_TOOLCHAIN "${POCKET_VULKAN_ROOT}/host-toolchain.cmake" CACHE FILEPATH "" FORCE)
    include_directories("${POCKET_VULKAN_ROOT}/Vulkan-Hpp" "${POCKET_VULKAN_ROOT}/spirv-install/include")
else()
    set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory(../../../../vendor/llama.cpp llama-build)
add_library(pocket_llama SHARED pocket_llama.cpp)
target_link_libraries(pocket_llama PRIVATE llama llama-common android log)
target_link_options(pocket_llama PRIVATE "-Wl,-z,max-page-size=16384")
